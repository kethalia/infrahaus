  <action>
Replace the entire `docker-compose.yml` with the following production-ready compose file. The current file only has a broken `web` service — replace it entirely.

Key ordering rules:
1. `postgres` and `redis` start first (no dependencies)
2. `migrate` waits for `postgres: service_healthy` then runs `prisma migrate deploy` and exits
3. `app` and `worker` both wait for `migrate: service_completed_successfully` AND `redis: service_healthy`

The `migrate` service uses `condition: service_completed_successfully` — this means Compose will wait for it to exit with code 0 before starting dependent services. This is the correct pattern to prevent running migrations while the app is already serving traffic.

Do NOT run migrations inside the app CMD (race condition). Do NOT use `restart: unless-stopped` on the migrate service (it must exit cleanly).

The worker service uses `build.target: worker-runner` to select the worker-runner stage from the multi-stage Dockerfile.

```yaml
services:
  postgres:
    image: postgres:16-alpine
    restart: unless-stopped
    environment:
      POSTGRES_USER: dashboard
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD}
      POSTGRES_DB: dashboard
    volumes:
      - postgres-data:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U dashboard"]
      interval: 5s
      timeout: 5s
      retries: 5

  redis:
    image: redis:7-alpine
    restart: unless-stopped
    command: redis-server --appendonly yes
    volumes:
      - redis-data:/data
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 5s
      timeout: 3s
      retries: 5

  migrate:
    build:
      context: .
      dockerfile: Dockerfile
      target: builder
    command: ["pnpm", "--filter", "dashboard", "exec", "prisma", "migrate", "deploy"]
    environment:
      DATABASE_URL: postgresql://dashboard:${POSTGRES_PASSWORD}@postgres:5432/dashboard
    depends_on:
      postgres:
        condition: service_healthy
    restart: "no"

  app:
    build:
      context: .
      dockerfile: Dockerfile
    restart: unless-stopped
    ports:
      - "${PORT:-3000}:3000"
    environment:
      DATABASE_URL: postgresql://dashboard:${POSTGRES_PASSWORD}@postgres:5432/dashboard
      REDIS_URL: redis://redis:6379
      ENCRYPTION_KEY: ${ENCRYPTION_KEY}
      SESSION_SECRET: ${SESSION_SECRET}
      NEXT_PUBLIC_WALLETCONNECT_PROJECT_ID: ${NEXT_PUBLIC_WALLETCONNECT_PROJECT_ID}
      HOSTNAME: "0.0.0.0"
    depends_on:
      migrate:
        condition: service_completed_successfully
      redis:
        condition: service_healthy

  worker:
    build:
      context: .
      dockerfile: Dockerfile
      target: worker-runner
    restart: unless-stopped
    environment:
      DATABASE_URL: postgresql://dashboard:${POSTGRES_PASSWORD}@postgres:5432/dashboard
      REDIS_URL: redis://redis:6379
      ENCRYPTION_KEY: ${ENCRYPTION_KEY}
    depends_on:
      migrate:
        condition: service_completed_successfully
      redis:
        condition: service_healthy

volumes:
  postgres-data:
  redis-data:
```

Write this to `docker-compose.yml` at the monorepo root, replacing the existing file.
